Por qué las aplicaciones necesitan contratos deterministas para agentes

Cuando un agente de IA interactúa con una aplicación mediante capturas de pantalla y clics por coordenadas (x, y), sufre de alta latencia, fragilidad ante cambios visuales de CSS y ausencia de fronteras de seguridad: una instrucción maliciosa dentro de un campo de texto o imagen puede inducir al agente a presionar un botón de transferencia bancaria.

Para que un agente opere tu aplicación con seguridad de grado financiero, debes exponer tus capacidades sobre cuatro pilares complementarios:

Pilar Estándar / API Qué construirás en el Sandbox
1. Herramientas Semánticas + Pausa Humana MCP (Model Context Protocol) Catálogo de 5 herramientas con ToolAnnotations y pausa elicitation_required (HTTP 202) para mutaciones operativas.
2. Descubrimiento entre Agentes A2A 1.0 (Agent2Agent) Manifiesto público en GET /.well-known/agent-card.json que declara la habilidad pericial de evaluación de daños.
3. Herramientas en el Navegador W3C WebMCP Registro imperativo en el DOM mediante navigator.modelContext.registerTool() reutilizando la sesión web activa.
4. Compuerta Criptográfica en Hardware ECDSA P-256 + SHA-256 Firma de args_hash en el dispositivo del asegurado y verificación determinista con tabla anti-replay consumedJti.
Axioma de seguridad para agentes con capacidad financiera Ningún prompt del sistema puede impedir al 100% una inyección indirecta. Las operaciones irreversibles (Nivel Rojo, como desembolsar un finiquito) deben exigir una prueba matemática firmada por la llave privada del dispositivo del usuario (Passkey / Keystore) atada exactamente al hash SHA-256 de los argumentos (args_hash).

Por qué usamos ECDSA P-256 (prime256v1) A diferencia de un secreto compartido HMAC, con criptografía asimétrica ECDSA P-256 el servidor de la aseguradora únicamente almacena la llave pública (SPKI PEM). La llave privada jamás abandona el chip seguro del teléfono o navegador del asegurado.

5. Compuerta Criptográfica Determinista ECDSA P-256

Duración estimada: 10 minutos

Llegamos al corazón de seguridad del codelab: proteger la herramienta de Nivel Rojo approve_settlement mediante tres funciones deterministas en demo-app/server.mjs:

  1. computeArgsHash(args): Ordena canónicamente las llaves { amountGtq, claimId, recipient } y calcula su digest SHA-256 en hexadecimal.
  2. signSettlementChallenge(args): Construye el sobre con un identificador único jti, adjunta el args_hash y lo firma con la llave privada EC P-256.
  3. verifySettlementGate(args, signatureToken, jtiStore): Verifica en el servidor la firma matemática con la llave pública EC P-256, comprueba que el args_hash firmado coincida byte por byte con los argumentos recibidos y registra el jti para impedir cobros duplicados.
demo-app/server.mjs (Parte 4 — Compuerta Criptográfica ECDSA P-256)
export function canonicalizeSettlementArgs(args) {
  const normalized = {
    amountGtq: Number(args.amountGtq),
    claimId: String(args.claimId || "").trim(),
    recipient: String(args.recipient || "").trim(),
  };
  return JSON.stringify(normalized);
}

export function computeArgsHash(args) {
  const canonicalJson = canonicalizeSettlementArgs(args);
  return crypto.createHash("sha256").update(canonicalJson, "utf8").digest("hex");
}

export function signSettlementChallenge(args, options = {}) {
  const privateKey = options.privateKey || DEVICE_KEYPAIR.privateKey;
  const jti = options.jti || `jti-${crypto.randomUUID().slice(0, 12)}`;
  const exp = options.exp ?? Date.now() + 5 * 60 * 1000;
  const args_hash = computeArgsHash(args);

  const payload = {
    jti,
    claimId: String(args.claimId),
    amountGtq: Number(args.amountGtq),
    recipient: String(args.recipient),
    args_hash,
    iat: Date.now(),
    exp,
    alg: "ES256",
  };

  const payloadB64Url = Buffer.from(JSON.stringify(payload), "utf8").toString("base64url");
  const signatureBuffer = crypto.sign("sha256", Buffer.from(payloadB64Url, "utf8"), privateKey);
  const signatureB64Url = signatureBuffer.toString("base64url");

  return {
    token: `${payloadB64Url}.${signatureB64Url}`,
    jti,
    args_hash,
    exp,
    canonicalJson: canonicalizeSettlementArgs(args),
    algorithm: "ES256 (ECDSA P-256 + SHA-256)",
  };
}

export function verifySettlementGate(args, signatureToken, jtiStore = consumedJti, publicKey = DEVICE_KEYPAIR.publicKey) {
  const computedArgsHash = computeArgsHash(args);

  if (!signatureToken || typeof signatureToken !== "string" || !signatureToken.includes(".")) {
    return {
      allowed: false,
      httpStatus: 403,
      code: "MISSING_SIGNATURE",
      computedArgsHash,
      message: "Compuerta Roja bloqueada: approve_settlement exige token firmado con ECDSA P-256 por el dispositivo del asegurado.",
    };
  }

  const [payloadB64Url, signatureB64Url] = signatureToken.split(".");
  const payload = JSON.parse(Buffer.from(payloadB64Url, "base64url").toString("utf8"));

  const signatureValid = crypto.verify(
    "sha256",
    Buffer.from(payloadB64Url, "utf8"),
    publicKey,
    Buffer.from(signatureB64Url, "base64url")
  );
  if (!signatureValid) {
    return { allowed: false, httpStatus: 403, code: "INVALID_ECDSA_SIGNATURE", computedArgsHash };
  }

  if (payload.args_hash !== computedArgsHash) {
    return {
      allowed: false,
      httpStatus: 403,
      code: "ARGS_HASH_MISMATCH",
      computedArgsHash,
      signedArgsHash: payload.args_hash,
      message: `Alerta de Tampering: el usuario firmó Q${payload.amountGtq}, pero el agente intentó ejecutar Q${args.amountGtq}.`,
    };
  }

  if (jtiStore.has(payload.jti)) {
    return {
      allowed: false,
      httpStatus: 403,
      code: "JTI_ALREADY_CONSUMED",
      jti: payload.jti,
      computedArgsHash,
      message: `Ataque de Replay neutralizado: el nonce '${payload.jti}' ya fue consumido.`,
    };
  }

  jtiStore.add(payload.jti);
  const disbursementId = `DISB-GT-${payload.jti.slice(-6).toUpperCase()}`;
  return {
    allowed: true,
    httpStatus: 200,
    code: "SETTLEMENT_DISBURSED",
    jti: payload.jti,
    computedArgsHash,
    disbursementId,
  };
}

6. Consola Web Interactiva y Laboratorio Red-Team en Vivo

Duración estimada: 10 minutos

Con el servidor listo, levantamos Escudo Agent Sandbox y ponemos a prueba la compuerta criptográfica desde el Laboratorio Red-Team integrado en la tercera columna de public/index.html.

Los 4 escenarios que ejecutas en el Laboratorio Red-Team

Escenario Qué intenta el atacante / agente Respuesta del Servidor
Ataque 1: Prompt Injection Una instrucción oculta en la foto del siniestro ordena llamar a approve_settlement por Q25,000 sin pasar por el dispositivo del asegurado. 403 MISSING_SIGNATURE
Ataque 2: Parameter Tampering El usuario firma legítimamente Q3,400, pero el agente comprometido altera el argumento amountGtq a Q15,000 conservando el token firmado. 403 ARGS_HASH_MISMATCH
Ataque 3: Replay Attack Un intermediario intercepta un token firmado válido y lo envía por segunda vez para duplicar la transferencia bancaria. 403 JTI_ALREADY_CONSUMED
4. Ejecución Legítima P-256 El dispositivo firma el desafío exacto (Q3,400 para el taller certificado) con un jti fresco y args_hash idéntico. 200 SETTLEMENT_DISBURSED

7. Verificación automatizada con node:test (gate.test.mjs)

Duración estimada: 6 minutos

Para garantizar que ninguna modificación futura debilite las pausas de Elicitación ni la compuerta criptográfica ECDSA P-256, escribimos la suite de pruebas automatizadas en demo-app/gate.test.mjs utilizando node:test:

demo-app/gate.test.mjs
import test from "node:test";
import assert from "node:assert/strict";
import {
  MCP_TOOLS,
  A2A_AGENT_CARD,
  computeArgsHash,
  signSettlementChallenge,
  verifySettlementGate,
  executeMcpTool,
} from "./server.mjs";

const sampleSettlementArgs = {
  claimId: "CLM-2026-089",
  amountGtq: 3400,
  recipient: "Taller Multimarcas Zona 10 (Cuenta GT82-ESCUDO-99)",
};

test("1. Catálogo MCP clasifica correctamente los 3 niveles de autonomía (Verde / Ámbar / Rojo) y Descriptor A2A 1.0", () => {
  const greenTools = MCP_TOOLS.filter(t => t.autonomyTier === "GREEN");
  const amberTools = MCP_TOOLS.filter(t => t.autonomyTier === "AMBER");
  const redTools = MCP_TOOLS.filter(t => t.autonomyTier === "RED");

  assert.equal(greenTools.length, 2);
  assert.equal(amberTools.length, 2);
  assert.equal(redTools.length, 1);
  assert.equal(greenTools.every(t => t.annotations.readOnlyHint === true), true);
  assert.equal(redTools[0].name, "approve_settlement");
  assert.equal(redTools[0].annotations.destructiveHint, true);
  assert.equal(A2A_AGENT_CARD.protocolVersion, "1.0");
});

test("2. Herramientas Ámbar devuelven pausa elicitation_required hasta recibir confirmación explícita del usuario", () => {
  const paused = executeMcpTool("assign_workshop", {
    claimId: "CLM-2026-089",
    workshopId: "WS-Z10-01",
    dropOffDate: "2026-10-12",
  }, { elicitationConfirmed: false });

  assert.equal(paused.httpStatus, 202);
  assert.equal(paused.status, "elicitation_required");
  assert.equal(paused.autonomyTier, "AMBER");

  const confirmed = executeMcpTool("assign_workshop", {
    claimId: "CLM-2026-089",
    workshopId: "WS-Z10-01",
    dropOffDate: "2026-10-12",
  }, { elicitationConfirmed: true });

  assert.equal(confirmed.httpStatus, 200);
  assert.equal(confirmed.status, "confirmed");
});

test("3. Ataque 1 (Prompt Injection sin firma en hardware) es bloqueado con 403 MISSING_SIGNATURE", () => {
  const verdict = verifySettlementGate({
    claimId: "CLM-2026-089",
    amountGtq: 25000,
    recipient: "Cuenta Atacante GT00-9999",
  }, null, new Set());

  assert.equal(verdict.allowed, false);
  assert.equal(verdict.httpStatus, 403);
  assert.equal(verdict.code, "MISSING_SIGNATURE");
});

test("4. Ataque 2 (Tampering de Q3,400 a Q15,000 tras firmar) es bloqueado con 403 ARGS_HASH_MISMATCH", () => {
  const signed = signSettlementChallenge(sampleSettlementArgs);
  const tamperedArgs = { ...sampleSettlementArgs, amountGtq: 15000 };

  const verdict = verifySettlementGate(tamperedArgs, signed.token, new Set());
  assert.equal(verdict.allowed, false);
  assert.equal(verdict.httpStatus, 403);
  assert.equal(verdict.code, "ARGS_HASH_MISMATCH");
});

test("5. Ataque 3 (Replay Attack reenviando el mismo JTI dos veces) aprueba el primero y bloquea el segundo con 403 JTI_ALREADY_CONSUMED", () => {
  const localJtiStore = new Set();
  const signed = signSettlementChallenge(sampleSettlementArgs);

  const firstAttempt = verifySettlementGate(sampleSettlementArgs, signed.token, localJtiStore);
  assert.equal(firstAttempt.allowed, true);
  assert.equal(firstAttempt.httpStatus, 200);
  assert.equal(firstAttempt.code, "SETTLEMENT_DISBURSED");

  const replayAttempt = verifySettlementGate(sampleSettlementArgs, signed.token, localJtiStore);
  assert.equal(replayAttempt.allowed, false);
  assert.equal(replayAttempt.httpStatus, 403);
  assert.equal(replayAttempt.code, "JTI_ALREADY_CONSUMED");
});

test("6. computeArgsHash produce un digest SHA-256 determinista independiente del orden de propiedades de entrada", () => {
  const h1 = computeArgsHash({
    claimId: "CLM-2026-089",
    amountGtq: 3400,
    recipient: "Taller Multimarcas Zona 10 (Cuenta GT82-ESCUDO-99)",
  });
  const h2 = computeArgsHash({
    recipient: "Taller Multimarcas Zona 10 (Cuenta GT82-ESCUDO-99)",
    amountGtq: 3400,
    claimId: "CLM-2026-089",
  });
  assert.equal(h1, h2);
  assert.equal(h1.length, 64);
});
